cmake_minimum_required(VERSION 3.16)
project(kf_toy VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kf STATIC src/filter.cpp)
target_include_directories(kf PUBLIC include)
target_compile_options(kf PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(replay src/replay.cpp)
target_link_libraries(replay PRIVATE kf)
target_compile_options(replay PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME replay_regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/verify.py $<TARGET_FILE:replay> ${CMAKE_BINARY_DIR}/verification)


find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(kf3d STATIC src/kf3d/geometry.cpp src/kf3d/propagation.cpp src/kf3d/filter.cpp)
target_include_directories(kf3d PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(kf3d PUBLIC Eigen3::Eigen)
target_compile_options(kf3d PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
set_target_properties(kf3d PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(replay3d src/replay3d.cpp vendor/utm/utm.cc vendor/utm/tranmerc.cc)
target_include_directories(replay3d PRIVATE vendor/utm)
target_link_libraries(replay3d PRIVATE kf3d)
target_compile_options(replay3d PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(kf_math_tests tests/math_contracts.cpp)
target_link_libraries(kf_math_tests PRIVATE kf3d)
add_test(NAME math_contracts COMMAND kf_math_tests)
add_executable(kf_teaching examples/teaching.cpp)
target_link_libraries(kf_teaching PRIVATE kf3d)
install(TARGETS kf3d EXPORT kf3dTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/kf3d DESTINATION include)
install(EXPORT kf3dTargets NAMESPACE kf3d:: DESTINATION lib/cmake/kf3d)
install(FILES cmake/kf3dConfig.cmake DESTINATION lib/cmake/kf3d)
add_test(NAME data_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/test_data_pipeline.py)
add_test(NAME replay_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_replay_protocol.py $<TARGET_FILE:replay3d>)
add_executable(typed_client examples/typed_client.cpp)
target_link_libraries(typed_client PRIVATE kf3d)
add_test(NAME typed_client COMMAND typed_client)
add_test(NAME teaching_determinism COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/verify_teaching.py $<TARGET_FILE:kf_teaching>)
add_executable(kf_public_api_tests tests/public_api.cpp)
target_link_libraries(kf_public_api_tests PRIVATE kf3d)
add_test(NAME public_api COMMAND kf_public_api_tests)
add_executable(kf_custom_components_tests tests/custom_components.cpp)
target_link_libraries(kf_custom_components_tests PRIVATE kf3d)
add_test(NAME custom_components COMMAND kf_custom_components_tests)
